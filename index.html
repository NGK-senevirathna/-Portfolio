TimetableController.cs — conflict check C# / ASP.NET Core 
bool IsSlotTaken(string hall, string day, int slotId)
{
    return _db.TimetableEntries.Any(e =>
        e.HallName == hall &&
        e.DayOfWeek == day &&
        e.SlotID == slotId);
}

// Called before every insert so two lecturers
// can never be booked into the same hall/slot.
if (IsSlotTaken(model.HallName, model.DayOfWeek, model.SlotID))
{
    ModelState.AddModelError("", "This hall is already booked for that slot.");
    return View(model);
}

private void btnLogin_Click(object sender, EventArgs e)
{
    string user = txtUsername.Text.Trim();
    string pass = txtPassword.Text;

    using var cmd = new SqlCommand(
        "SELECT COUNT(*) FROM Users WHERE Username=@u AND PasswordHash=@p",
        conn);
    cmd.Parameters.AddWithValue("@u", user);
    cmd.Parameters.AddWithValue("@p", Hash(pass));

    if ((int)cmd.ExecuteScalar() == 1)
        new Dashboard().Show();
